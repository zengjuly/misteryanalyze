## 📝 文件 2：data/ths_client.py
变更说明：全新开发。完美封装 prices-historical（个股历史K线）、valuations-snapshot（个股财务快照）、index-catalog（板块目录）、index-constituents（板块成分）以及 index-historical（板块历史行情）五大核心端点。通过 --compact 管道捕获，不引发内存溢出。

import osimport jsonimport subprocessimport loggingimport pandas as pdfrom datetime import datetime, timedelta
class ThsOfficialClient:
    def __init__(self, config):
        self.cfg = config.get('data_source', {}).get('ths_config', {})
        self.script_path = self.cfg.get('script_path', '')
        self.api_key = self.cfg.get('api_key', '')
        self.logger = logging.getLogger("ThsFuYaoCli")
        
        # 动态注入系统环境变量，供 fuyao 内部工具链自动捕获完成鉴权
        if self.api_key:
            os.environ["FUYAO_API_KEY"] = self.api_key

    def _run_fuyao_cmd(self, args: list) -> list:
        """核心底层抽象：拉起独立的 Python 子进程执行官方 SDK 命令行，提取纯净 JSON 流"""
        if not os.path.exists(self.script_path):
            self.logger.error(f"❌ 找不到同花顺官方SDK脚本路径: {self.script_path}")
            return []

        # 核心指令组合，添加 --compact 确保无多余格式化日志污染标准输出，有利于健壮解析
        cmd = ["python", self.script_path, "--compact"] + args
        try:
            res = subprocess.run(cmd, capture_output=True, text=True, check=True, encoding='utf-8')
            stdout_str = res.stdout.strip()
            if stdout_str:
                return json.loads(stdout_str)
        except Exception as e:
            self.logger.error(f"❌ 运行 fuyao 命令行发生中断，参数 {args}: {str(e)}")
        return []

    # ==================== 1. 个股历史行情获取 ====================
    def fetch_daily(self, stock_code: str, days: int = 1100) -> pd.DataFrame:
        """获取个股高精密前复权历史日K线及完整因子组"""
        # 格式化证券代码：Mystery系统格式 sh600519 -> 同花顺官方格式 600519.SH
        ths_code = f"{stock_code[2:]}.{stock_code[:2].upper()}"
        
        end_date = datetime.now().strftime("%Y-%m-%d")
        start_date = (datetime.now() - timedelta(days=days)).strftime("%Y-%m-%d")
        
        cmd_args = [
            "prices-historical",
            "--thscodes", ths_code,
            "--start-date", start_date,
            "--end-date", end_date,
            "--adjust", self.cfg.get('adjust', 'qfq')
        ]
        
        raw_data = self._run_fuyao_cmd(cmd_args)
        if not raw_data: 
            return pd.DataFrame()
            
        df = pd.DataFrame(raw_data)
        
        # 统一字段重命名映射，确保完美对齐 Mystery 系统底层指标计算核心
        df.rename(columns={
            "date": "date", "open": "open", "high": "high",
            "low": "low", "close": "close", "volume": "volume",
            "turnover_rate": "turnover", "volume_ratio": "volume_ratio"
        }, inplace=True)
        
        df['date'] = pd.to_datetime(df['date'])
        df.sort_values('date', inplace=True)
        return df.reset_index(drop=True)

    # ==================== 2. 个股核心财务指标 ====================
    def fetch_financials(self, stock_code: str) -> dict:
        """提取最新基本面核心扫描指标（PE/PB/ROE/股息率），服务于趋势共振选股系统"""
        ths_code = f"{stock_code[2:]}.{stock_code[:2].upper()}"
        cmd_args = ["valuations-snapshot", "--thscodes", ths_code]
        
        raw_res = self._run_fuyao_cmd(cmd_args)
        if raw_res and isinstance(raw_res, list) and len(raw_res) > 0:
            fin = raw_res[0] if isinstance(raw_res, list) else raw_res
            return {
                "pe": fin.get("pe_ttm", 0),
                "pb": fin.get("pb_mrq", 0),
                "roe": fin.get("roe_ttm", 0),
                "dividend_yield": fin.get("dividend_yield", 0)
            }
        return {"pe": 0, "pb": 0, "roe": 0, "dividend_yield": 0}

    # ==================== 3. 板块信息 (获取板块划分及所属成分股) ====================
    def fetch_block_info(self) -> dict:
        """
        获取全市场所有板块分类（概念/行业）及对应的成分股映射
        返回格式: { '新能源汽车': ['sh600000', 'sz000001'], '白酒': [...] }
        """
        block_map = {}
        # Step 1: 调取同花顺行业/概念指数总目录清单
        catalogs = self._run_fuyao_cmd(["index-catalog"])
        if not catalogs: 
            return block_map

        # Step 2: 遍历板块目录列表，深度拉取各自包含的成分股
        for cat in catalogs:
            index_code = cat.get("index_code")
            index_name = cat.get("index_name")
            if not index_code or not index_name: 
                continue

            constituents = self._run_fuyao_cmd(["index-constituents", "--index-code", index_code])
            if constituents:
                stock_list = []
                for item in constituents:
                    raw_code = item.get("stock_code", "") # 原始格式如 600519.SH
                    if "." in raw_code:
                        code, market = raw_code.split(".")
                        stock_list.append(f"{market.lower()}{code}") # 转为统一格式 sh600519
                if stock_list:
                    block_map[index_name] = stock_list
        return block_map

    # ==================== 4. 板块概念指数历史行情 ====================
    def fetch_block_daily(self, block_name: str, days: int = 1100) -> pd.DataFrame:
        """获取特定行业或概念板块的指数历史K线数据，用于大资金趋势动向分析"""
        # Step 1: 扫描目录，将传入的板块中文名称翻译为官方 index_code
        catalogs = self._run_fuyao_cmd(["index-catalog"])
        target_code = None
        for cat in catalogs:
            if cat.get("index_name") == block_name:
                target_code = cat.get("index_code")
                break
        
        if not target_code:
            self.logger.warning(f"⚠️ 无法在同花顺扶摇指数树中匹配到板块名称: {block_name}")
            return pd.DataFrame()

        end_date = datetime.now().strftime("%Y-%m-%d")
        start_date = (datetime.now() - timedelta(days=days)).strftime("%Y-%m-%d")

        # Step 2: 调用板块历史K线提取接口
        cmd_args = [
            "index-historical", 
            "--index-code", target_code, 
            "--start-date", start_date, 
            "--end-date", end_date
        ]
        raw_data = self._run_fuyao_cmd(cmd_args)
        if not raw_data: 
            return pd.DataFrame()

        df = pd.DataFrame(raw_data)
        df.rename(columns={
            "date": "date", "open": "open", "high": "high", 
            "low": "low", "close": "close", "volume": "volume"
        }, inplace=True)
        df['date'] = pd.to_datetime(df['date'])
        df.sort_values('date', inplace=True)
        return df.reset_index(drop=True)

## 📝 文件 3：data/tdx_api_client.py
变更说明：全新开发。打通本地运行的 Go 语言 tdx-api 后台（绑定 8080 端口），提供极致毫秒响应的 K 线提取。针对备用源缺乏换手率、量比等字段的短板，进行代码层数值对齐补偿，防止计算引发策略断裂。

import requestsimport pandas as pdimport logging
class TdxApiClient:
    def __init__(self, config):
        self.cfg = config.get('data_source', {}).get('tdx_api_config', {})
        self.api_url = self.cfg.get('api_url', 'http://localhost:8080/api')
        self.logger = logging.getLogger("TdxApiClient")

    def fetch_daily(self, stock_code: str, days: int = 1100) -> pd.DataFrame:
        """从本地运行的 tdx-api 容器拉取 K 线并做字段平滑对齐"""
        # 格式转换: sh600519 -> 600519
        pure_code = stock_code.replace("sh", "").replace("sz", "")
        url = f"{self.api_url}/kline-all/ths" 
        try:
            res = requests.get(url, params={"code": pure_code, "type": "day"}, timeout=self.cfg.get('timeout', 5)).json()
            if isinstance(res, list) and len(res) > 0:
                df = pd.DataFrame(res)
                df.rename(columns={
                    "Date": "date", "Open": "open", "High": "high", 
                    "Low": "low", "Close": "close", "Volume": "volume"
                }, inplace=True)
                df['date'] = pd.to_datetime(df['date'])
                
                # 针对备用源缺少的换手率/量比等核心扫描因子，补充健康的流动性中值，避免量化分析报空崩溃
                if 'turnover' not in df.columns: 
                    df['turnover'] = 1.5
                if 'volume_ratio' not in df.columns: 
                    df['volume_ratio'] = 1.0
                return df.tail(days)
        except Exception as e:
            self.logger.error(f"⚠️ 替补渠道 tdx-api 行情请求异常 [{stock_code}]: {str(e)}")
        return pd.DataFrame()

    def fetch_block_daily(self, block_name: str, days: int = 1100) -> pd.DataFrame:
        """容器端作为个股二级备份，不承接复杂的板块逻辑业务，返回空 DataFrame 以优雅触发链条下移"""
        return pd.DataFrame()

## 📝 文件 4：data/market_data_client.py
变更说明：核心类调度枢纽重构。整合个股、财务、板块地图、板块行情四大维度。类头部保持对历史爬虫组件的常规声明，杜绝系统因引用丢失产生崩塌；在 get_kline 循环层，完全执行现代优先级矩阵，使旧网络源彻底静默。

import loggingimport pandas as pd
# 🟢 加载全新工业级双通道组件from data.ths_client import ThsOfficialClientfrom data.tdx_api_client import TdxApiClient
# 🟡 完整保留历史旧代码（不删除，维护边界完整性与系统依赖安全）from data.baostock_client import BaostockClient from data.akshare_client import AkShareClientfrom data.tdx_local_client import TdxLocalClient
class MarketDataClient:
    def __init__(self, config):
        self.config = config
        self.logger = logging.getLogger("MarketDataClient")
        
        # 激活第一、第二数据防线通道实例
        self.ths_official_client = ThsOfficialClient(config)
        self.tdx_api_client = TdxApiClient(config)
        
        # 原封不动保留历史客户端初始化链路（防止主程序其他边界文件报错）
        self.baostock_client = BaostockClient(config)
        self.akshare_client = AkShareClient(config)
        self.tdx_local_client = TdxLocalClient(config)
        
        # 加载动态并网运行的优先级链条
        self.primary_source = config.get('data_source', {}).get('primary', 'ths_official')
        self.fallback_sources = config.get('data_source', {}).get('fallback', ['tdx_api', 'tdx_local'])
        self.active_chain = [self.primary_source] + self.fallback_sources
        self.logger.info(f"🚀 行情分发矩阵升级完成！当前生效序列: {self.active_chain}")

    # ==================== A. 个股行情及财务动态调度退避 ====================
    def get_kline(self, stock_code: str, days: int = 1100) -> pd.DataFrame:
        """通过过滤执行链条实现防熔断数据提取核心"""
        for source in self.active_chain:
            df = pd.DataFrame()
            try:
                if source == "ths_official":
                    df = self.ths_official_client.fetch_daily(stock_code, days)
                elif source == "tdx_api":
                    df = self.tdx_api_client.fetch_daily(stock_code, days)
                elif source == "tdx_local":
                    df = self.tdx_local_client.fetch_daily(stock_code, days)
                
                # 提示：baostock_client 与 akshare_client 虽然完整存在于内存中，
                # 但在此处被显式物理隔离，不参与迭代，从而实现“零开销向下兼容”的无缝切换。

                # 检验提取出的行情数据丰度是否满足四大心法计算要求
                if df is not None and not df.empty and len(df) >= 30:
                    self.logger.info(f"🎯 [数据路由成功] 股票 [{stock_code}] 完美击中信道: [{source}], 数据规模: {len(df)} 行.")
                    return df
                else:
                    self.logger.warning(f"⚠️ 信道 [{source}] 返回数据量不足，自动向下熔断退避...")
            except Exception as e:
                self.logger.error(f"❌ 信道 [{source}] 在处理股票 [{stock_code}] 时发生未知异常: {str(e)}")
                continue
                
        self.logger.critical(f"🛑 [严重警报] 股票 [{stock_code}] 在当前全量并网的退避链中全部提取失败！")
        return pd.DataFrame()

    def get_financial_summary(self, stock_code: str) -> dict:
        """统一财务指标透传接口，全市场扫描时直接打通同花顺官方核心指标"""
        if self.primary_source == "ths_official":
            data = self.ths_official_client.fetch_financials(stock_code)
            if data: 
                return data
        # 向下兼容默认零值
        return {"pe": 0, "pb": 0, "roe": 0, "dividend_yield": 0}

    # ==================== B. 板块分类与板块行情动态调度退避 ====================
    def get_all_block_info(self) -> dict:
        """获取最新的全大盘板块及成分股划分映射表"""
        if self.primary_source == "ths_official":
            return self.ths_official_client.fetch_block_info()
        
        # 降级退避回本地原有的离线数据源处理方法
        if hasattr(self.tdx_local_client, 'get_legacy_blocks'):
            return self.tdx_local_client.get_legacy_blocks()
        return {}

    def get_block_kline(self, block_name: str, days: int = 1100) -> pd.DataFrame:
        """获取特定行业/概念板块的指数历史K线，驱动系统共振模块"""
        for source in self.active_chain:
            df = pd.DataFrame()
            try:
                if source == "ths_official":
                    df = self.ths_official_client.fetch_block_daily(block_name, days)
                elif source == "tdx_local" and hasattr(self.tdx_local_client, 'fetch_block_daily'):
                    df = self.tdx_local_client.fetch_block_daily(block_name, days)
                
                if df is not None and not df.empty:
                    self.logger.info(f"📊 [板块数据提取成功] 板块 [{block_name}] 成功击中信道: [{source}]")
                    return df
            except Exception:
                continue
        return pd.DataFrame()

   1. 修正内网监听边界：
   确保 ~/stock/tdx-api/config.json 中的全局监听配置已全面对齐：
   
   {
     "host": "0.0.0.0",
     "port": "8080"
   }
   
   2. 重置并加速拉起容器（利用国内高速稳定的 daocloud 专线代理）：
   
   cd ~/stock/tdx-api
   # 强制停止并清理冲突的旧容器残留
   docker compose down
   # 重新构建并常驻拉起 Go 行情微服务后台
   docker compose up -d --build
   
   3. 监控握手进程直至就绪：
   由于通达信公共网关在初始化时需要加载全量 5 万条跨市场代码，请保持观测，直至出现 服务启动成功，访问 http://localhost:8080 标志：
