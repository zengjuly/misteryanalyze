### 2.2 新增方法（插入在 `fetch_block_info` 之前或之后）

```python
def get_index_catalog(self, force: bool = False) -> list:
    """index-catalog 一次拉取，进程内缓存。返回 [{thscode, name}, ...]"""
    if self._index_catalog_cache is not None and not force:
        return self._index_catalog_cache
    raw = self._run_fuyao(['index-catalog'])
    out = []
    name_map = {}
    for cat in (raw or []):
        if not isinstance(cat, dict):
            continue
        code = cat.get('thscode') or cat.get('index_code')
        name = cat.get('name') or cat.get('index_name')
        if not code or not name:
            continue
        out.append({'thscode': str(code), 'name': str(name)})
        name_map[str(name)] = str(code)
    self._index_catalog_cache = out
    self._index_name_to_code = name_map
    return out

def fetch_block_daily_by_code(self, thscode: str, days: int = 1100,
                              start_date: str = None,
                              end_date: str = None) -> pd.DataFrame:
    """板块指数日K：index-historical --thscode，不扫目录。"""
    if not thscode:
        return pd.DataFrame()
    if end_date is None:
        end = datetime.now()
    else:
        end = pd.to_datetime(end_date)
    if start_date is None:
        start = end - timedelta(days=days)
    else:
        start = pd.to_datetime(start_date)
    raw = self._run_fuyao([
        'index-historical', '--thscode', str(thscode),
        '--start-ms', str(int(start.timestamp() * 1000)),
        '--end-ms', str(int(end.timestamp() * 1000))])
    if not raw:
        return pd.DataFrame()
    df = pd.DataFrame(raw)
    if df.empty or 'date_ms' not in df.columns:
        return pd.DataFrame()
    out = pd.DataFrame()
    out['日期'] = pd.to_datetime(df['date_ms'], unit='ms')
    out['开盘价'] = df.get('open_price', 0).astype(float)
    out['最高价'] = df.get('high_price', 0).astype(float)
    out['最低价'] = df.get('low_price', 0).astype(float)
    out['收盘价'] = df.get('close_price', 0).astype(float)
    out['成交量'] = df.get('volume', 0).astype(float)
    out['成交额'] = df.get('turnover', 0).astype(float)
    out['换手率'] = None
    return out.sort_values('日期').reset_index(drop=True)
```

### 2.3 改写 `fetch_block_daily`

用缓存目录解析名称，再调 `fetch_block_daily_by_code`，删除内部每次 `index-catalog` 全量扫描逻辑：

```python
def fetch_block_daily(self, block_name: str, days: int = 1100,
                      start_date: str = None,
                      end_date: str = None) -> pd.DataFrame:
    self.get_index_catalog()
    target = self._index_name_to_code.get(block_name)
    if not target:
        for cat in (self._index_catalog_cache or []):
            if cat.get('name') == block_name:
                target = cat.get('thscode')
                break
    if not target:
        return pd.DataFrame()
    return self.fetch_block_daily_by_code(
        target, days=days, start_date=start_date, end_date=end_date)
```

        except Exception as e:

            continue
    return pd.DataFrame()

## 4. `web/pages_util.py` —— 全量重写 `calc_sector_strength`

**删除**所有 `db.load_kline(c, ...)` 对成分股循环、`codes[:10]`、`样本数` 逻辑。

用下面实现整文件替换核心函数（保留 import / logger / `_strength_cache` / `_CACHE_TTL` / `build_sector_strength_map`）：

```python
def calc_sector_strength(use_cache: bool = True) -> list:
    """板块强度：Financial-API 板块指数日K，禁止个股抽样/全量。
    得分 = MA20偏离×0.4 + 近10日涨幅×0.3 + 成交额放大×0.3
    """
    import time
    if use_cache and _strength_cache['rows'] \
            and time.time() - _strength_cache['ts'] < _CACHE_TTL:
        return _strength_cache['rows']
    try:
        import yaml
        import numpy as np
        root = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
        cfg_path = os.path.join(root, 'config', 'config.yaml')
        cfg = yaml.safe_load(open(cfg_path, encoding='utf-8')) or {}
        from data.ths_client import ThsOfficialClient
        # 兼容从项目根导入
        try:
            from ths_client import ThsOfficialClient as _C
            ThsOfficialClient = _C
        except Exception:
            pass
        client = ThsOfficialClient(cfg)
        catalog = client.get_index_catalog()
        if not catalog:
            logger.warning('⚠️ index-catalog 为空，板块强度无法计算')
            return []
        rows = []
        # 限制数量：性能上限（可配置）；默认最多 120 个板块
        max_sectors = int((cfg.get('sector') or {}).get('max_sectors', 120))
        for cat in catalog[:max_sectors]:
            name = cat.get('name')
            code = cat.get('thscode')
            if not name or not code:
                continue
            try:
                kdf = client.fetch_block_daily_by_code(code, days=90)
                if kdf is None or len(kdf) < 20:
                    continue
                close = kdf['收盘价'].astype(float)
                last = float(close.iloc[-1])
                ma20 = float(close.tail(20).mean())
                if ma20 <= 0:
                    continue
                bias = (last / ma20 - 1) * 100
                chg10 = 0.0
                if len(close) > 11:
                    past = float(close.iloc[-11])
                    if past > 0:
                        chg10 = (last / past - 1) * 100
                amt_ratio = 1.0
                if '成交额' in kdf.columns:
                    a5 = float(kdf['成交额'].tail(5).mean() or 0)
                    a15 = float(kdf['成交额'].tail(20).head(15).mean() or 0)
                    if a15 > 0:
                        amt_ratio = a5 / a15
                score = bias * 0.4 + chg10 * 0.3 + (amt_ratio - 1) * 100 * 0.3
                rows.append({
                    '板块': name,
                    '板块代码': code,
                    'MA20偏离%': round(bias, 2),
                    '近10日涨幅%': round(chg10, 2),
                    '成交额放大': round(amt_ratio, 2),
                    '板块得分': round(float(score), 2),
                })
            except Exception:
                continue
        rows.sort(key=lambda r: r['板块得分'], reverse=True)
        _strength_cache['ts'] = time.time()
        _strength_cache['rows'] = rows
        return rows
    except Exception as e:
        logger.warning(f'⚠️ 板块强度计算失败: {e}')
        return []


def build_sector_strength_map() -> dict:
    return {r['板块']: r['板块得分'] for r in calc_sector_strength()}
```

导入失败时：`sys.path` 已含项目根与 `data/`，优先 `from ths_client import ThsOfficialClient`。
