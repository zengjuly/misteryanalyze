#!/usr/bin/env python3
ma_indicators.py - 均线系统技术指标计算（已扩展 MA377/610 + EMA）
import pandas as pd
import numpy as np
import logging
from typing import List, Optional

class MAIndicators:
    """均线系统技术指标计算"""

    def init(self):
        self.logger = logging.getLogger(name)

    def calculate_ma(self, data: pd.DataFrame, periods: List[int] = None) -> pd.DataFrame:
        """
        计算均线
        默认包含 MA5/10/20/60/250/377/610
        """
        if periods is None:
            periods = [5, 10, 20, 60, 250, 377, 610]

        try:
            result = data.copy()
            close_col = '收盘价' if '收盘价' in result.columns else 'close'
            for period in periods:
                ma_col = f'MA{period}'
                result[ma_col] = result[close_col].rolling(window=period, min_periods=1).mean()
            self.logger.info(f"完成 {len(periods)} 条均线计算: {periods}")
            return result
        except Exception as e:
            self.logger.error(f"计算均线异常: {e}")
            return data

    def calculate_ema(self, data: pd.DataFrame, periods: List[int] = None) -> pd.DataFrame:
        """计算 EMA（默认 EMA20）"""
        if periods is None:
            periods = [20]
        try:
            result = data.copy()
            close_col = '收盘价' if '收盘价' in result.columns else 'close'
            for p in periods:
                result[f'EMA{p}'] = result[close_col].ewm(span=p, adjust=False).mean()
            self.logger.info(f"完成 EMA 计算: {periods}")
            return result
        except Exception as e:
            self.logger.error(f"计算 EMA 异常: {e}")
            return data

#!/usr/bin/env python3
tdx_block_client.py - 通达信行业/概念/风格板块读取（复用 mootdx）
import os
import logging
from typing import Dict, List, Optional
import pandas as pd

logger = logging.getLogger(name)

class TdxBlockClient:
    def init(self, tdx_dir: str = None):
        self.tdx_dir = tdx_dir or os.environ.get('TDX_HOME') or os.environ.get('TDX_VIPDOC_DIR')
        self.reader = None
        if self.tdx_dir and os.path.exists(self.tdx_dir):
            try:
                from mootdx.reader import Reader
                self.reader = Reader.factory(market='std', tdxdir=self.tdx_dir)
                logger.info(f"TDX Block Client 初始化成功: {self.tdx_dir}")
            except Exception as e:
                logger.warning(f"mootdx Reader 初始化失败: {e}")

    def get_industry_blocks(self, group: bool = True) -> pd.DataFrame:
        """获取通达信行业板块（优先 block_gn.dat / incon.dat）"""
        if not self.reader:
            return pd.DataFrame()
        try:
概念/行业常用
            df = self.reader.block(symbol='block_gn.dat', group=group)
            if df is None or df.empty:
                df = self.reader.block(symbol='block.dat', group=group)
            return df if df is not None else pd.DataFrame()
        except Exception as e:
            logger.error(f"读取行业板块失败: {e}")
            return pd.DataFrame()

    def get_all_blocks(self) -> Dict[str, pd.DataFrame]:
        """返回各类板块"""
        result = {}
        for name, file in [
            ('概念', 'block_gn.dat'),
            ('风格', 'block_fg.dat'),
            ('指数', 'block_zs.dat'),
            ('行业', 'block.dat'),
        ]:
            try:
                df = self.reader.block(symbol=file, group=True)
                if df is not None and not df.empty:
                    result[name] = df
            except Exception:
                pass
        return result

    def get_stock_industry(self, code: str) -> Optional[str]:
        """单只股票所属行业（简单反查）"""
        df = self.get_industry_blocks()
        if df.empty:
            return None
根据实际列名调整（blockname / code）
        matched = df[df['code'].astype(str).str.contains(code[-6:])]
        if not matched.empty:
            return matched.iloc[0].get('blockname') or matched.iloc[0].get('name')
        return None

                )
            ''')

    def add(self, code: str, name: str = '', source: str = 'manual', note: str = ''):
        with sqlite3.connect(self.db_path) as conn:
            conn.execute(
                'INSERT OR REPLACE INTO watchlist (code, name, add_time, source, note) VALUES (?, ?, ?, ?, ?)',
                (code, name, datetime.now().isoformat(), source, note)
            )

    def remove(self, code: str):
        with sqlite3.connect(self.db_path) as conn:
            conn.execute('DELETE FROM watchlist WHERE code=?', (code,))

    def list_all(self) -> pd.DataFrame:
        with sqlite3.connect(self.db_path) as conn:
            return pd.read_sql_query('SELECT * FROM watchlist ORDER BY add_time DESC', conn)

    def exists(self, code: str) -> bool:
        with sqlite3.connect(self.db_path) as conn:
            cur = conn.execute('SELECT 1 FROM watchlist WHERE code=?', (code,))
            return cur.fetchone() is not None

import threading
import uuid
from data.db_manager import DEFAULT_DB_PATH  # 假设已有

def _update_job(conn, job_id, **kwargs):
简单更新 status/progress/result
    sets = ', '.join(f"{k}=?" for k in kwargs)
    conn.execute(f"UPDATE scan_jobs SET {sets} WHERE job_id=?", list(kwargs.values()) + [job_id])

def run_market_scan_background(limit=None, period='daily', sync_first=False, top_n=20):
    """后台运行，返回 job_id"""
    job_id = str(uuid.uuid4())
    with sqlite3.connect(DEFAULT_DB_PATH) as conn:
        conn.execute('''
            CREATE TABLE IF NOT EXISTS scan_jobs (
                job_id TEXT PRIMARY KEY,
                status TEXT,
                progress REAL,
                result_path TEXT,
                start_time TEXT,
                end_time TEXT,
                message TEXT
            )
        ''')
        conn.execute(
            'INSERT INTO scan_jobs (job_id, status, progress, start_time) VALUES (?, ?, ?, ?)',
            (job_id, 'running', 0.0, datetime.now().isoformat())
        )

    def _worker():
        try:
            result = run_market_scan(limit=limit, period=period, sync_first=sync_first, top_n=top_n)
            with sqlite3.connect(DEFAULT_DB_PATH) as conn:
                _update_job(conn, job_id,
                            status='finished',
                            progress=1.0,
                            result_path=result.get('明细路径', ''),
                            end_time=datetime.now().isoformat(),
                            message=f"完成，信号数 {result.get('含信号数', 0)}")
        except Exception as e:
            with sqlite3.connect(DEFAULT_DB_PATH) as conn:
                _update_job(conn, job_id, status='failed', message=str(e))

    t = threading.Thread(target=_worker, daemon=True)
    t.start()
    return job_id
