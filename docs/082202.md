#!/usr/bin/env python3# -*- coding: utf-8 -*-"""
Mystery 趋势交易分析系统 - 数据库架构升级脚本
功能：初始化并升级 SQLite 行业板块指数行情表，支持同花顺(Financial-API)与通达信(tdxapi)双源。"""
import osimport sqlite3import logging
# 配置日志输出
logging.basicConfig(
    level=logging.INFO,
    format='%(asctime)s - %(levelname)s - [%(filename)s:%(lineno)d] - %(message)s'
)logger = logging.getLogger("MysteryDBUpgrade")
def upgrade_database(db_path: str):
    """执行数据库升级事务，创建板块标准存储及索引"""
    
    # 确保数据库目录存在
    db_dir = os.path.dirname(db_path)
    if db_dir and not os.path.exists(db_dir):
        os.makedirs(db_dir)
        logger.info(f"创建数据库目录: {db_dir}")

    conn = None
    try:
        # 连接数据库（若不存在则自动创建）
        conn = sqlite3.connect(db_path)
        cursor = conn.cursor()
        
        # 开启显式事务
        cursor.execute("BEGIN TRANSACTION;")
        
        logger.info(f"正在对数据库进行升级变更: {db_path}")

        # 1. 创建板块行情主表 (sector_kline)
        # 支持字段：开高低收、成交量、成交额，以及用于追溯的数据源标签
        cursor.execute("""
            CREATE TABLE IF NOT EXISTS sector_kline (
                sector_code TEXT NOT NULL,      -- 板块/行业唯一代码 (例如: ths_881155, tdx_880301)
                sector_name TEXT NOT NULL,      -- 板块/行业标准名称 (例如: 半导体, 证券)
                trade_date TEXT NOT NULL,       -- 交易日期 (格式: YYYY-MM-DD)
                open REAL NOT NULL,             -- 开盘价
                high REAL NOT NULL,             -- 最高价
                low REAL NOT NULL,              -- 最低价
                close REAL NOT NULL,            -- 收盘价
                volume INTEGER NOT NULL,        -- 成交量 (手/股)
                amount REAL NOT NULL,           -- 成交额 (元)
                source_type TEXT DEFAULT 'ths', -- 数据源标签: 'ths'(同花顺扶摇) 或 'tdx'(通达信)
                update_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP, -- 记录写入时间
                PRIMARY KEY (sector_code, trade_date)
            );
        """)
        logger.info("✅ 行业板块指数行情表 `sector_kline` 初始化/校验成功。")

        # 2. 创建复合性能索引
        # 索引覆盖原则：加速依据交易日期筛选全市场板块，以及加速单个板块的时间序列分析
        cursor.execute("""
            CREATE INDEX IF NOT EXISTS idx_sector_kline_date 
            ON sector_kline (trade_date, sector_code);
        """)
        
        cursor.execute("""
            CREATE INDEX IF NOT EXISTS idx_sector_kline_code_date 
            ON sector_kline (sector_code, trade_date DESC);
        """)
        logger.info("✅ 板块行级别高性能复合索引创建成功。")

        # 3. 创建板块元数据资产表 (sector_meta)
        # 用于存储板块分类、成分股映射状态，避免每次扫描去解析大 JSON
        cursor.execute("""
            CREATE TABLE IF NOT EXISTS sector_meta (
                sector_code TEXT PRIMARY KEY,
                sector_name TEXT NOT NULL,
                parent_type TEXT,               -- 板块分类: '行业'、'概念'、'申万一级'
                base_code TEXT,                 -- 原始官方代码 (如: 881155)
                is_active INTEGER DEFAULT 1,    -- 是否在监控观察池 (1: 活跃, 0: 弃用)
                last_sync_date TEXT             -- 上次成功增量同步行情到哪一天 (YYYY-MM-DD)
            );
        """)
        logger.info("✅ 行业板块元数据资产表 `sector_meta` 初始化/校验成功。")

        # 提交事务
        conn.commit()
        logger.info("🎉 Mystery 系统 SQLite 行业指数数据层升级圆满完成。")

    except sqlite3.Error as e:
        if conn:
            conn.rollback()
            logger.error(f"❌ 数据库升级失败，事务已回滚。错误摘要: {str(e)}")
        raise e
    finally:
        if conn:
            conn.close()

#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
Mystery 趋势交易分析系统 - 行业共振层精准纠偏
功能：彻底切断个股反向取样计算，改由真实板块指数进行 MA 偏离与多头共振评估
"""

import sqlite3
import numpy as np

class ResonanceAnalyzer:
    def __init__(self, db_path: str):
        self.db_path = db_path

    def calculate_industry_score(self, sector_code: str) -> float:
        """
        根据真实行业板块指数计算行业趋势分（满分 25 分）
        公式规整：MA20偏离(40%) + 近10日涨幅(30%) + 成交额放大(30%)
        """
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()
        
        # 1. 逆序秒杀获取行业指数最近 60 个交易日的收盘价与成交额（利用索引高效提取）
        cursor.execute("""
            SELECT close, amount FROM sector_kline 
            WHERE sector_code = ? 
            ORDER BY trade_date DESC LIMIT 60;
        """, (sector_code,))
        rows = cursor.fetchall()
        conn.close()
        
        if len(rows) < 20:
            return 12.5  # 数据不足返回中位风险基准分

        # 倒序转换成时间序列正序 [过去 -> 现在]
        closes = np.array([row[0] for row in rows])[::-1]
        amounts = np.array([row[1] for row in rows])[::-1]
        
        current_close = closes[-1]
        ma20 = np.mean(closes[-20:])
        
        # 指标 A: MA20 偏离度分项 (Max: 10分)
        bias_20 = (current_close - ma20) / ma20
        bias_score = min(10.0, max(0.0, (bias_20 * 100) + 5.0))
        
        # 指标 B: 近 10 日真实涨幅分项 (Max: 7.5分)
        return_10d = (closes[-1] - closes[-10]) / closes[-10]
        return_score = min(7.5, max(0.0, (return_10d * 100) + 3.75))
        
        # 指标 C: 板块成交额多日量能对比分项 (Max: 7.5分)
        recent_amount = np.mean(amounts[-5:])
        history_amount = np.mean(amounts[-20:])
        volume_ratio = recent_amount / (history_amount + 1e-6)
        volume_score = min(7.5, max(0.0, volume_ratio * 3.75))
        
        # 聚合行业维度的真实得分
        total_industry_score = bias_score + return_score + volume_score
        return float(round(total_industry_score, 2))
